## Tasks

### Task 1: 安装 Vitest 和配置

**Step 1: 安装 Vitest 依赖**
```bash
npm install -D vitest @vue/test-utils @vitest/coverage-v8 happy-dom
```

**Step 2: 创建 vitest.config.js**
```javascript
import { defineConfig } from 'vitest/config'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  test: {
    environment: 'happy-dom',
    globals: true,
    coverage: {
      provider: 'v8',
      reporter: ['text', 'html'],
      exclude: ['node_modules/', 'scripts/', 'dist/']
    }
  }
})
```

**Step 3: 添加测试脚本到 package.json**
```json
{
  "scripts": {
    "test": "vitest",
    "test:run": "vitest run",
    "test:coverage": "vitest run --coverage"
  }
}
```

**Step 4: 提交配置**
```bash
git add vitest.config.js package.json package-lock.json
git commit -m "chore: add Vitest testing framework"
```

### Task 2: 编写核心工具函数测试

**Step 1: 创建测试目录**
```bash
mkdir -p src/utils/__tests__
```

**Step 2: 测试 spacedRepetition.js**
创建 `src/utils/__tests__/spacedRepetition.test.js`:
```javascript
import { describe, it, expect } from 'vitest'
import { calculateNextReview, getIntervalLevel } from '../spacedRepetition.js'

describe('spacedRepetition', () => {
  describe('calculateNextReview', () => {
    it('should return a timestamp in the future', () => {
      const result = calculateNextReview(0)
      expect(result).toBeGreaterThan(Date.now())
    })

    it('should increase interval with higher level', () => {
      const level0 = calculateNextReview(0)
      const level1 = calculateNextReview(1)
      expect(level1).toBeGreaterThan(level0)
    })

    it('should respect ease factor', () => {
      const normal = calculateNextReview(1, 2.5)
      const harder = calculateNextReview(1, 1.5)
      expect(normal).toBeGreaterThan(harder)
    })
  })
})
```

**Step 3: 测试 vocabularyManager.js**
创建 `src/utils/__tests__/vocabularyManager.test.js`:
```javascript
import { describe, it, expect } from 'vitest'
import { VOCABULARIES, getVocabById, getVocabByLevel } from '../vocabularyManager.js'

describe('vocabularyManager', () => {
  describe('VOCABULARIES', () => {
    it('should be an array', () => {
      expect(Array.isArray(VOCABULARIES)).toBe(true)
    })

    it('should have valid structure', () => {
      VOCABULARIES.forEach(v => {
        expect(v).toHaveProperty('id')
        expect(v).toHaveProperty('name')
        expect(v).toHaveProperty('file')
      })
    })
  })

  describe('getVocabById', () => {
    it('should return vocab by id', () => {
      const vocab = getVocabById('vocab-cet4-basic')
      expect(vocab).toBeDefined()
      expect(vocab.id).toBe('vocab-cet4-basic')
    })

    it('should return undefined for unknown id', () => {
      const vocab = getVocabById('unknown-vocab')
      expect(vocab).toBeUndefined()
    })
  })
})
```

### Task 3: 配置 ESLint

**Step 1: 安装 ESLint 依赖**
```bash
npm install -D eslint @eslint/js eslint-plugin-vue
```

**Step 2: 创建 eslint.config.js**
```javascript
import js from '@eslint/js'
import pluginVue from 'eslint-plugin-vue'

export default [
  {
    ignores: ['dist/', 'node_modules/', '.planning/']
  },
  js.configs.recommended,
  ...pluginVue.configs['flat/recommended'],
  {
    files: ['**/*.{js,vue}'],
    rules: {
      'vue/multi-word-component-names': 'off',
      'no-unused-vars': 'warn',
      'no-console': 'warn'
    }
  }
]
```

**Step 3: 添加 lint 脚本**
```json
{
  "scripts": {
    "lint": "eslint .",
    "lint:fix": "eslint . --fix"
  }
}
```

**Step 4: 运行 lint**
```bash
npm run lint
```

**Step 5: 提交配置**
```bash
git add eslint.config.js package.json package-lock.json
git commit -m "chore: add ESLint configuration"
```
